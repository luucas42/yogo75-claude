```swift
        XCTAssertEqual(Set(Command.allCases.map(\.rawValue)), [0x10, 0x11, 0x3C, 0x3D])
        XCTAssertNil(Command(rawValue: 0x3B), "flash write must be unrepresentable")

    func testHeaderLayout() {
        let p = Packet.encode(.startSession, seq: 0x1234)
        XCTAssertEqual(p.count, 64)
        XCTAssertEqual(Array(p[0..<8]), [0xAA, 0x10, 0x00, 0x00, 0x00, 0x34, 0x12, 0x00])
        XCTAssertTrue(p[8...].allSatisfy { $0 == 0 })
    }

    func testRGB565() {
        XCTAssertEqual(RGB(255, 0, 0).rgb565LE, [0x00, 0xF8])
        XCTAssertEqual(RGB(0, 255, 0).rgb565LE, [0xE0, 0x07])
        XCTAssertEqual(RGB(0, 0, 255).rgb565LE, [0x1F, 0x00])
        XCTAssertEqual(RGB(255, 255, 255).rgb565LE, [0xFF, 0xFF])

    func testHexRoundTrip() throws {
        XCTAssertEqual(RGB(hex: "e2794f"), RGB(0xE2, 0x79, 0x4F))
        XCTAssertEqual(RGB(0xE2, 0x79, 0x4F).hex, "e2794f")
        XCTAssertNil(RGB(hex: "+e2794"))
        XCTAssertNil(RGB(hex: "e2794"))

        XCTAssertEqual(String(decoding: data, as: UTF8.self), #"["010203"]"#)
        XCTAssertEqual(try JSONDecoder().decode([RGB].self, from: data), [RGB(1, 2, 3)])
    }

    /// Golden bytes: exactly what the 2026-10-14 spike sent and the user saw as a full red screen.
    func testFullRedFrameMatchesSpike() {
        let r = Frame.filled(RGB(255, 0, 0)).liveReports()
        XCTAssertEqual(r.count, 2)
        XCTAssertEqual(Array(r[0][0..<8]), [0xAA, 0x3C, 0x00, 0x00, 0x38, 0x00, 0x00, 0x00])
        XCTAssertEqual(Array(r[0][8..<64]), Array(repeating: [0x00, 0xF8], count: 28).flatMap { $0 })

        XCTAssertTrue(r[1][24...].allSatisfy { $0 == 0 })

    func testCornerPixelPlacement() {
        var f = Frame.blank
```

```swift
import Foundation

/// One LED colour. Encodes to JSON as a 6-digit lowercase hex string ("e2794f").
public struct RGB: Hashable, Sendable {
    public var r: UInt8
    public var g: UInt8
    public var b: UInt8

    public init(_ r: UInt8, _ g: UInt8, _ b: UInt8) {
        self.r = r
        self.g = g
        self.b = b
    }

    public static let off = RGB(0, 0, 0)

    public init?(hex: String) {
        guard hex.count == 6, hex.allSatisfy(\.isHexDigit), let v = UInt32(hex, radix: 16) else { return nil }
        self.init(UInt8((v >> 16) & 0xFF), UInt8((v >> 8) & 0xFF), UInt8(v & 0xFF))
    }

    public var hex: String { String(format: "%02x%02x%02x", r, g, b) }

    /// RGB565 in little-endian byte order, as the 0x3C live-frame command expects.
    public var rgb565LE: [UInt8] {
        let v = UInt16(r >> 3) << 11 | UInt16(g >> 2) << 5 | UInt16(b >> 3)
        return [UInt8(v & 0xFF), UInt8(v >> 8)]
    }
}

extension RGB: Codable {
    public init(from decoder: Decoder) throws {
        let c = try decoder.singleValueContainer()
        let s = try c.decode(String.self)
        guard let v = RGB(hex: s) else {
            throw DecodingError.dataCorruptedError(in: c, debugDescription: "expected 6 hex digits, got \(s)")
        }
        self = v
    }

    public func encode(to encoder: Encoder) throws {
        var c = encoder.singleValueContainer()
        try c.encode(hex)
    }
}
```

```swift
    case endSession = 0x11
    case liveFrame = 0x3C
    case endLive = 0x3D
}
```

`Sources/YogoCore/Protocol/Packet.swift`:

```swift
/// 64-byte output report: AA <cmd> <offset lo> <offset hi> <len> <seq lo> <seq hi> 00, payload from byte 8.
public enum Packet {
    public static let size = 64
    public static let headerSize = 8
    public static let maxPayload = size - headerSize

    public static func encode(_ cmd: Command, offset: UInt16 = 0, data: [UInt8] = [], seq: UInt16 = 0) -> [UInt8] {
        precondition(data.count <= maxPayload, "payload larger than \(maxPayload) bytes")
        var p = [UInt8](repeating: 0, count: size)
        p[0] = 0xAA
        p[1] = cmd.rawValue
        p[2] = UInt8(offset & 0xFF)
        p[3] = UInt8(offset >> 8)
        p[4] = UInt8(data.count)
        p[5] = UInt8(seq & 0xFF)
        p[6] = UInt8(seq >> 8)
        p.replaceSubrange(headerSize..<(headerSize + data.count), with: data)
        return p
    }

    /// A reply whose byte 2 is 0xFF means the firmware rejected the command.
    public static func isRejection(_ reply: [UInt8]) -> Bool {
        reply.count >= headerSize && reply[0] == 0xAA && reply[2] == 0xFF
    }
}
```

`Sources/YogoCore/Protocol/Frame.swift`:

```swift
/// One 6×6 screen image, row-major from the top-left (0 = top-left, 5 = top-right, 30 = bottom-left).
public struct Frame: Hashable, Sendable {
    public static let width = 6
    public static let height = 6
    public static let pixelCount = width * height

    public private(set) var pixels: [RGB]

    public init(pixels: [RGB]) {
        precondition(pixels.count == Frame.pixelCount, "a frame has exactly 36 pixels")
        self.pixels = pixels
    }

    public static let blank = Frame(pixels: Array(repeating: .off, count: pixelCount))

    public static func filled(_ c: RGB) -> Frame { Frame(pixels: Array(repeating: c, count: pixelCount)) }

    public subscript(x: Int, y: Int) -> RGB {
        get { pixels[y * Frame.width + x] }
        set { pixels[y * Frame.width + x] = newValue }
    }

    /// The two 0x3C reports for this frame: RGB565 bytes 0..<56 at offset 0, 56..<72 at offset 56.
    public func liveReports() -> [[UInt8]] {
        let bytes = pixels.flatMap(\.rgb565LE)
        return [
            Packet.encode(.liveFrame, offset: 0, data: Array(bytes[0..<56])),
            Packet.encode(.liveFrame, offset: 56, data: Array(bytes[56..<72])),
        ]
    }
}
```

```swift
import Foundation

public struct AnimationFrame: Codable, Hashable, Sendable {
    public var durationMs: Int
    public var pixels: [RGB]

    public init(durationMs: Int, frame: Frame) {
        self.durationMs = durationMs
        self.pixels = frame.pixels
    }

    /// Only valid after `PixelAnimation.validate()` has passed (36 pixels).
    public var frame: Frame { Frame(pixels: pixels) }
}

/// A 6×6 animation, stored as `<id>.json` (spec §7). Named PixelAnimation to avoid SwiftUI.Animation.
public struct PixelAnimation: Codable, Hashable, Identifiable, Sendable {
    public static let formatVersion = 1
    public static let maxFrames = 20
    public static let durationRange = 50...5000

    public var version: Int
    public var id: String
    public var name: String
    public var loop: Bool
    public var frames: [AnimationFrame]

    public init(id: String, name: String, loop: Bool, frames: [AnimationFrame]) {
        self.version = PixelAnimation.formatVersion
        self.id = id
        self.name = name
        self.loop = loop
        self.frames = frames
    }

    public enum ValidationError: Error, Equatable {
        case unsupportedVersion(Int)
        case emptyID
        case noFrames
        case tooManyFrames(Int)
        case badPixelCount(frame: Int, count: Int)
        case badDuration(frame: Int, ms: Int)
    }

    public func validate() throws {
        guard version == PixelAnimation.formatVersion else { throw ValidationError.unsupportedVersion(version) }
        guard !id.isEmpty else { throw ValidationError.emptyID }
        guard !frames.isEmpty else { throw ValidationError.noFrames }
        guard frames.count <= PixelAnimation.maxFrames else { throw ValidationError.tooManyFrames(frames.count) }
        for (i, f) in frames.enumerated() {
            guard f.pixels.count == Frame.pixelCount else {
                throw ValidationError.badPixelCount(frame: i, count: f.pixels.count)
            }
            guard PixelAnimation.durationRange.contains(f.durationMs) else {
                throw ValidationError.badDuration(frame: i, ms: f.durationMs)
            }
        }
    }

    public static func clampDuration(_ ms: Int) -> Int {
        min(max(ms, durationRange.lowerBound), durationRange.upperBound)
    }

    public static func decode(_ data: Data) throws -> PixelAnimation {
        let a = try JSONDecoder().decode(PixelAnimation.self, from: data)
        try a.validate()
        return a
    }

    public func encoded() throws -> Data {
        let e = JSONEncoder()
        e.outputFormatting = [.prettyPrinted, .sortedKeys]
        return try e.encode(self)
    }
```

```swift
import Foundation

/// One Claude Code hook event as forwarded by `yogo hook` (spec §4).
public struct SignalMessage: Codable, Equatable, Sendable {
    public var event: String
    public var session: String
    public var notificationType: String?
    public var tool: String?
    public var toolUseID: String?

    public init(event: String, session: String, notificationType: String? = nil,
                tool: String? = nil, toolUseID: String? = nil) {
        self.event = event
        self.session = session
        self.notificationType = notificationType
        self.tool = tool
        self.toolUseID = toolUseID
    }

    /// Parses Claude Code's hook stdin JSON. Returns nil for anything unusable; the caller must stay silent.
    /// No size cap: PostToolUse payloads can be large, and dropping one would leave a tool "running".
    public static func fromHookPayload(_ data: Data) -> SignalMessage? {
        guard let obj = (try? JSONSerialization.jsonObject(with: data)) as? [String: Any],
              let event = obj["hook_event_name"] as? String, !event.isEmpty,
              let session = obj["session_id"] as? String, !session.isEmpty
        else { return nil }
        return SignalMessage(event: event, session: session,
                             notificationType: obj["notification_type"] as? String,
                             tool: obj["tool_name"] as? String,
                             toolUseID: obj["tool_use_id"] as? String)
    }

    /// Synthetic events for `yogo simulate <state>`.
    public static func simulate(_ state: ClaudeState, session: String) -> [SignalMessage] {
        switch state {
        case .thinking: return [SignalMessage(event: "UserPromptSubmit", session: session)]
        case .tool: return [SignalMessage(event: "PreToolUse", session: session, tool: "Bash", toolUseID: "simulate")]
        case .waiting: return [SignalMessage(event: "PermissionRequest", session: session)]
        case .done: return [SignalMessage(event: "Stop", session: session)]
        case .idle: return [SignalMessage(event: "SessionEnd", session: session)]
        }
    }
}
```

```swift
import Foundation

/// Per-session Claude Code state machine (spec §5). Not thread-safe: use it from one queue.
public final class ClaudeTracker {
    public struct Config: Equatable {
        public var staleTimeout: TimeInterval = 600
        public var waitingTimeout: TimeInterval = 7200
        /// Length of the assigned "done" animation; the app keeps this in sync (built-in: 1.35 s).
        public var doneDuration: TimeInterval = 1.35
        public init() {}
    }

    struct Session {
        enum Phase { case idle, active, done }
        var phase: Phase = .idle
        var running: Set<String> = []
        var waitingOn: Set<String> = []
        var lastEventAt: Date
        var doneAt: Date?

        var rawState: ClaudeState {
            if !waitingOn.isEmpty { return .waiting }
            switch phase {
            case .idle: return .idle
            case .done: return .done
            case .active: return running.isEmpty ? .thinking : .tool
            }
        }
    }

    static let unknownID = "?"
    static let waitingNotifications: Set<String> = ["permission_prompt", "elicitation_dialog", "agent_needs_input"]
    static let userBlockingTools: Set<String> = ["AskUserQuestion", "ExitPlanMode"]

    public var config: Config
    private(set) var sessions: [String: Session] = [:]

    public init(config: Config = Config()) { self.config = config }

    public var sessionCount: Int { sessions.count }

    public func reset() { sessions.removeAll() }

    public func apply(_ m: SignalMessage, at now: Date) {
        if m.event == "SessionEnd" {
            sessions[m.session] = nil
            return
        }
        var s = sessions[m.session] ?? Session(lastEventAt: now)
        s.lastEventAt = now
        let id = m.toolUseID ?? Self.unknownID
        switch m.event {
        case "UserPromptSubmit":
            s.phase = .active; s.running = []; s.waitingOn = []
        case "PreToolUse":
            // Claude moving on to a new tool means any earlier prompt was answered (or denied).
            s.phase = .active; s.waitingOn = []
            if let tool = m.tool, Self.userBlockingTools.contains(tool) { s.waitingOn.insert(id) } else { s.running.insert(id) }
        case "PostToolUse", "PostToolUseFailure":
            if m.toolUseID == nil {
                s.running = []; s.waitingOn = []
            } else {
                s.running.remove(id); s.waitingOn.remove(id); s.waitingOn.remove(Self.unknownID)
            }
        case "PermissionRequest":
            s.phase = .active; s.waitingOn.insert(id)
        case "Notification":
            guard let type = m.notificationType else { break }
            if Self.waitingNotifications.contains(type) {
                s.phase = .active; s.waitingOn.insert(Self.unknownID)
            } else if type == "idle_prompt", s.waitingOn.isEmpty {
                s.phase = .idle; s.running = []
            }
        case "Stop":
            s.phase = .done; s.doneAt = now; s.running = []; s.waitingOn = []
        case "StopFailure":
            s.phase = .idle; s.running = []; s.waitingOn = []
        default:
            break   // SessionStart (incl. source=compact) and unknown events only refresh lastEventAt
        }
        sessions[m.session] = s
    }

    public func state(at now: Date) -> ClaudeState {
        sessions.values.map { effectiveState($0, at: now) }.max() ?? .idle
    }

    func effectiveState(_ s: Session, at now: Date) -> ClaudeState {
        let quiet = now.timeIntervalSince(s.lastEventAt)
        switch s.rawState {
        case .thinking, .tool: return quiet >= config.staleTimeout ? .idle : s.rawState
        case .waiting: return quiet >= config.waitingTimeout ? .idle : .waiting
        case .done: return now.timeIntervalSince(s.doneAt ?? s.lastEventAt) >= config.doneDuration ? .idle : .done
        case .idle: return .idle
        }
    }
}
```

```swift
import Foundation

/// Maps (scene, time) to the frame to show. Durations snap to the 50 ms tick grid (spec §4).
public enum Player {
    public static let tickMs = 50

    public static func quantize(_ ms: Int) -> Int {
        max(tickMs, Int((Double(ms) / Double(tickMs)).rounded()) * tickMs)
    }

    public static func totalMs(_ a: PixelAnimation) -> Int {
        a.frames.reduce(0) { $0 + quantize($1.durationMs) }
    }

    public static func frame(for scene: ScreenScene, at now: Date) -> Frame? {
        guard case let .play(anim, since, loop) = scene, let last = anim.frames.last else { return nil }
        let total = totalMs(anim)
        var t = max(0, Int(now.timeIntervalSince(since) * 1000))
        if loop {
            t %= total
        } else if t >= total {
            return last.frame
        }
        for f in anim.frames {
            let d = quantize(f.durationMs)
            if t < d { return f.frame }
            t -= d
        }
        return last.frame
    }
}
```

```swift
import Foundation

public protocol ScreenDevice: AnyObject {
    var isOpen: Bool { get }
    /// Sends one 64-byte output report. Throws on I/O failure.
    func send(_ report: [UInt8]) throws
}

/// Streams frames to the keyboard (spec §4). Call `tick` at 20 Hz from a single queue.
public final class ScreenDriver {
    public static let sessionRefresh: TimeInterval = 2

    public private(set) var isLive = false
    public private(set) var consecutiveErrors = 0
    private var device: ScreenDevice?
    private var lastSessionAt: Date?
    private var seq: UInt16 = 0

    public init() {}

    /// A newly attached (or re-opened) device always starts without a session.
    public func attach(_ device: ScreenDevice?) {
        self.device = device
        isLive = false
        lastSessionAt = nil
        consecutiveErrors = 0
    }

    /// `frame == nil` gives the screen back to the keyboard (0x3D + 0x11, exactly once).
    public func tick(frame: Frame?, at now: Date) {
        guard let device, device.isOpen else { return }
        do {
            if let frame {
                if lastSessionAt.map({ now.timeIntervalSince($0) >= Self.sessionRefresh }) ?? true {
                    seq &+= 1
                    try device.send(Packet.encode(.startSession, seq: seq))
                    lastSessionAt = now
                }
                for report in frame.liveReports() { try device.send(report) }
                isLive = true
            } else if isLive {
                // Flip state first: if the device is gone, the firmware reverts on its own in 0.5 s.
                isLive = false
                lastSessionAt = nil
                try device.send(Packet.encode(.endLive))
                try device.send(Packet.encode(.endSession))
            }
            consecutiveErrors = 0
        } catch {
            consecutiveErrors += 1
        }
    }

    /// Clean shutdown before quit or sleep.
    public func release() { tick(frame: nil, at: Date()) }
}
```

```swift
import Foundation

/// Spec §11: five rejected replies in a row pause live mode for 30 s.
public struct RejectionGuard {
    public static let limit = 5
    public static let pause: TimeInterval = 30

    private var streak = 0
    public private(set) var pausedUntil: Date?

    public init() {}

    public mutating func record(reply: [UInt8], at now: Date) {
        guard Packet.isRejection(reply) else {
            streak = 0
            return
        }
        streak += 1
        if streak >= Self.limit {
            pausedUntil = now + Self.pause
            streak = 0
        }
    }

    public func isPaused(at now: Date) -> Bool { pausedUntil.map { now < $0 } ?? false }
}
```

```swift
/// Append-only log with simple rotation (spec §11: 1 MB × 3 files).
public final class Log {
    public static let shared = Log(file: YogoPaths.logFile)

    public let file: URL
    let limit: Int
    let keep: Int
    private let queue = DispatchQueue(label: "yogo.log")
    private let formatter = ISO8601DateFormatter()

    public init(file: URL, limit: Int = 1_000_000, keep: Int = 3) {
        self.file = file
        self.limit = limit
        self.keep = keep
    }

    public func info(_ message: String) { write("INFO", message) }
    public func warn(_ message: String) { write("WARN", message) }

    /// Blocks until queued lines are on disk.
    public func flush() { queue.sync {} }

    private func write(_ level: String, _ message: String) {
        let line = "\(formatter.string(from: Date())) \(level) \(message)\n"
        queue.async { [self] in
            try? FileManager.default.createDirectory(at: file.deletingLastPathComponent(), withIntermediateDirectories: true)
            rotateIfNeeded()
            if let h = try? FileHandle(forWritingTo: file) {
                h.seekToEndOfFile()
                h.write(Data(line.utf8))
                try? h.close()
            } else {
                try? Data(line.utf8).write(to: file)
            }
        }
    }

    func rotateIfNeeded() {
        let fm = FileManager.default
        guard let size = (try? fm.attributesOfItem(atPath: file.path))?[.size] as? Int, size >= limit else { return }
        let p = file.path
        try? fm.removeItem(atPath: "\(p).\(keep - 1)")
        if keep > 2 {
            for i in stride(from: keep - 2, through: 1, by: -1) {
                try? fm.moveItem(atPath: "\(p).\(i)", toPath: "\(p).\(i + 1)")
            }
        }
        try? fm.moveItem(atPath: p, toPath: "\(p).1")
    }
}
```

```swift
import Darwin
import Foundation

/// Minimal AF_UNIX stream helpers: one JSON line per connection.
public enum UnixSocket {
    public enum SocketError: Error, Equatable {
        case pathTooLong
        case system(String, Int32)
    }

    static func address(_ path: String) throws -> sockaddr_un {
        var addr = sockaddr_un()
        addr.sun_family = sa_family_t(AF_UNIX)
        let bytes = Array(path.utf8)
        guard bytes.count < MemoryLayout.size(ofValue: addr.sun_path) else { throw SocketError.pathTooLong }
        withUnsafeMutableBytes(of: &addr.sun_path) { raw in
            raw.copyBytes(from: bytes)
            raw[bytes.count] = 0
        }
        addr.sun_len = UInt8(MemoryLayout<sockaddr_un>.size)
        return addr
    }

    static func setTimeouts(_ fd: Int32, _ seconds: TimeInterval) {
        var tv = timeval(tv_sec: Int(seconds), tv_usec: Int32((seconds - seconds.rounded(.down)) * 1_000_000))
        setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &tv, socklen_t(MemoryLayout<timeval>.size))
        setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &tv, socklen_t(MemoryLayout<timeval>.size))
        var one: Int32 = 1
        setsockopt(fd, SOL_SOCKET, SO_NOSIGPIPE, &one, socklen_t(MemoryLayout<Int32>.size))
    }

    static func connected(_ path: String, timeout: TimeInterval) throws -> Int32 {
        var addr = try address(path)
        let fd = socket(AF_UNIX, SOCK_STREAM, 0)
        guard fd >= 0 else { throw SocketError.system("socket", errno) }
        setTimeouts(fd, timeout)
        let rc = withUnsafePointer(to: &addr) {
            $0.withMemoryRebound(to: sockaddr.self, capacity: 1) { connect(fd, $0, socklen_t(MemoryLayout<sockaddr_un>.size)) }
        }
        guard rc == 0 else {
            let e = errno
            close(fd)
            throw SocketError.system("connect", e)
        }
        return fd
    }

    public static func isListening(path: String) -> Bool {
        guard let fd = try? connected(path, timeout: 0.3) else { return false }
        close(fd)
        return true
    }

    /// Connects, writes `line` + "\n", and optionally reads one reply line, all within `timeout`.
    public static func request(path: String, line: Data, expectReply: Bool, timeout: TimeInterval) throws -> Data? {
        let fd = try connected(path, timeout: timeout)
        defer { close(fd) }
        try writeAll(fd, line + Data([0x0A]))
        return expectReply ? try readLine(fd) : nil
    }

    static func writeAll(_ fd: Int32, _ data: Data) throws {
        try data.withUnsafeBytes { (buf: UnsafeRawBufferPointer) in
            var off = 0
            while off < buf.count {
                let n = write(fd, buf.baseAddress! + off, buf.count - off)
                guard n > 0 else { throw SocketError.system("write", errno) }
                off += n
            }
        }
    }

    static func readLine(_ fd: Int32, limit: Int = 1 << 20) throws -> Data {
        var out = Data()
        var buf = [UInt8](repeating: 0, count: 4096)
        while out.count < limit {
            let n = read(fd, &buf, buf.count)
            if n < 0 { throw SocketError.system("read", errno) }
            if n == 0 { break }
            out.append(contentsOf: buf[0..<n])
            if let nl = out.firstIndex(of: 0x0A) { return out.subdata(in: out.startIndex..<nl) }
        }
        return out
    }
```

```swift
import Darwin
import Foundation

/// Accepts one-line requests on a Unix socket and hands them to `handler` on `handlerQueue`.
public final class SocketServer {
    public enum ServerError: Error, Equatable {
        case alreadyRunning
        case system(String, Int32)
    }

    public typealias Handler = (Data) -> Data?

    private let path: String
    private let handler: Handler
    private let handlerQueue: DispatchQueue
    private let ioQueue = DispatchQueue(label: "yogo.socket.io")
    private var listenFD: Int32 = -1
    private var source: DispatchSourceRead?

    public init(path: String, handlerQueue: DispatchQueue = .main, handler: @escaping Handler) {
        self.path = path
        self.handlerQueue = handlerQueue
        self.handler = handler
    }

    public func start() throws {
        // Spec §11: never take the socket (and with it the keyboard) from a live instance.
        if UnixSocket.isListening(path: path) { throw ServerError.alreadyRunning }
        try FileManager.default.createDirectory(at: URL(fileURLWithPath: path).deletingLastPathComponent(),
                                                withIntermediateDirectories: true)
        var addr = try UnixSocket.address(path)
        unlink(path)
        let fd = socket(AF_UNIX, SOCK_STREAM, 0)
        guard fd >= 0 else { throw ServerError.system("socket", errno) }
        let rc = withUnsafePointer(to: &addr) {
            $0.withMemoryRebound(to: sockaddr.self, capacity: 1) { bind(fd, $0, socklen_t(MemoryLayout<sockaddr_un>.size)) }
        }
        guard rc == 0 else {
            let e = errno
            close(fd)
            throw ServerError.system("bind", e)
        }
        chmod(path, 0o600)
        guard listen(fd, 16) == 0 else {
            let e = errno
            close(fd)
            throw ServerError.system("listen", e)
        }
        listenFD = fd
        let src = DispatchSource.makeReadSource(fileDescriptor: fd, queue: ioQueue)
        src.setEventHandler { [weak self] in self?.acceptOne() }
        src.setCancelHandler { close(fd) }
        src.resume()
        source = src
    }

    private func acceptOne() {
        let client = accept(listenFD, nil, nil)
        guard client >= 0 else { return }
        defer { close(client) }
        UnixSocket.setTimeouts(client, 1)
        guard let line = try? UnixSocket.readLine(client), !line.isEmpty else { return }
        let reply = handlerQueue.sync { handler(line) }
        if let reply { try? UnixSocket.writeAll(client, reply + Data([0x0A])) }
    }

    public func stop() {
        guard let src = source else { return }
        source = nil
        src.cancel()
        unlink(path)
    }

    deinit { stop() }
}
```

```swift
import Foundation

/// Built-ins (read-only, in code) plus user animations stored as `<id>.json` (spec §7).
public final class AnimationStore {
    public enum StoreError: Error, Equatable {
        case builtinIsReadOnly(String)
        case invalidID(String)
    }

    public let directory: URL
    public private(set) var user: [PixelAnimation] = []
    /// File names that were not loaded (corrupt, invalid, duplicate, or shadowing a built-in).
    public private(set) var skippedFiles: [String] = []

    public init(directory: URL) { self.directory = directory }

    public var all: [PixelAnimation] { BuiltinAnimations.all + user }

    public func get(_ id: String) -> PixelAnimation? { all.first { $0.id == id } }

    public static func isBuiltin(_ id: String) -> Bool { id.hasPrefix("builtin.") }

    static func isValidID(_ id: String) -> Bool {
        !id.isEmpty && id.allSatisfy { $0.isASCII && ($0.isLetter || $0.isNumber || "._-".contains($0)) } && !id.contains("..")
    }

    public func reload() {
        user = []
        skippedFiles = []
        let urls = (try? FileManager.default.contentsOfDirectory(at: directory, includingPropertiesForKeys: nil)) ?? []
        for url in urls.sorted(by: { $0.lastPathComponent < $1.lastPathComponent }) where url.pathExtension == "json" {
            guard let data = try? Data(contentsOf: url),
                  let a = try? PixelAnimation.decode(data),
                  !Self.isBuiltin(a.id), Self.isValidID(a.id),
                  !user.contains(where: { $0.id == a.id })
            else {
                skippedFiles.append(url.lastPathComponent)
                continue
            }
            user.append(a)
        }
        user.sort { $0.name.localizedStandardCompare($1.name) == .orderedAscending }
    }

    public func save(_ a: PixelAnimation) throws {
        guard !Self.isBuiltin(a.id) else { throw StoreError.builtinIsReadOnly(a.id) }
        guard Self.isValidID(a.id) else { throw StoreError.invalidID(a.id) }
        try a.validate()
        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        try a.encoded().write(to: fileURL(a.id), options: .atomic)
        if let i = user.firstIndex(where: { $0.id == a.id }) { user[i] = a } else { user.append(a) }
    }

    public func delete(_ id: String) throws {
        guard !Self.isBuiltin(id) else { throw StoreError.builtinIsReadOnly(id) }
        guard Self.isValidID(id) else { throw StoreError.invalidID(id) }
        try? FileManager.default.removeItem(at: fileURL(id))
        user.removeAll { $0.id == id }
    }

    /// A user-owned copy with a fresh id (used when a built-in is edited, spec §7).
    public static func copy(of a: PixelAnimation, name: String? = nil) -> PixelAnimation {
        var c = a
        c.id = "user." + UUID().uuidString.lowercased()
        c.name = name ?? "\(a.name) (kopya)"
        return c
    }

    public static func newBlank(name: String = "Yeni animasyon") -> PixelAnimation {
        PixelAnimation(id: "user." + UUID().uuidString.lowercased(), name: name, loop: true,
                       frames: [AnimationFrame(durationMs: 200, frame: .blank)])
    }

    func fileURL(_ id: String) -> URL { directory.appendingPathComponent("\(id).json") }
}

public enum Assignments {
    /// Resolves stored ids (ClaudeState.rawValue → animation id). Missing, unknown or deleted ids fall
    /// back to the built-in default (spec §11). `idle` never gets an animation.
    public static func resolve(_ ids: [String: String], lookup: (String) -> PixelAnimation?) -> [ClaudeState: PixelAnimation] {
        var out = BuiltinAnimations.defaultAssignments
        for (key, id) in ids {
            guard let state = ClaudeState(rawValue: key), state != .idle, let a = lookup(id) else { continue }
            out[state] = a
        }
        return out
    }
}
```

```swift
import Foundation

/// Merges Yogo's async hooks into Claude Code's settings.json without touching anything else (spec §10).
public enum HooksInstaller {
    public static let events = [
        "SessionStart", "UserPromptSubmit", "PreToolUse", "PostToolUse", "PostToolUseFailure",
        "PermissionRequest", "Notification", "Stop", "StopFailure", "SessionEnd",
    ]

    public enum InstallError: Error, Equatable {
        case invalidJSON
        case unexpectedShape(String)
    }

    static let pathMarker = "/Yogo.app/Contents/MacOS/yogo"

    public static func command(cliPath: String) -> String { "'\(cliPath)' hook" }

    static func isOurs(_ handler: Any) -> Bool {
        guard let cmd = (handler as? [String: Any])?["command"] as? String else { return false }
        return cmd.contains(pathMarker) && cmd.hasSuffix(" hook")
    }

    static func parse(_ data: Data) throws -> [String: Any] {
        if data.allSatisfy({ [0x20, 0x09, 0x0A, 0x0D].contains($0) }) { return [:] }
        guard let obj = try? JSONSerialization.jsonObject(with: data), let dict = obj as? [String: Any] else {
            throw InstallError.invalidJSON
        }
        return dict
    }

    static func serialize(_ dict: [String: Any]) throws -> Data {
        var d = try JSONSerialization.data(withJSONObject: dict, options: [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes])
        d.append(0x0A)
        return d
    }

    static func hooksDict(_ root: [String: Any]) throws -> [String: Any] {
        guard let h = root["hooks"] else { return [:] }
        guard let dict = h as? [String: Any] else { throw InstallError.unexpectedShape("hooks") }
        return dict
    }

    /// Removes our handlers everywhere; drops matcher groups and events that only contained ours.
    /// Events we did not touch keep their original value.
    static func strip(_ hooks: [String: Any]) -> [String: Any] {
        var out: [String: Any] = [:]
        for (event, value) in hooks {
            guard let groups = value as? [Any] else {
                out[event] = value
                continue
            }
            var changed = false
            var kept: [Any] = []
            for g in groups {
                guard var group = g as? [String: Any], let hs = group["hooks"] as? [Any] else {
                    kept.append(g)
                    continue
                }
                let remaining = hs.filter { !isOurs($0) }
                if remaining.count == hs.count {
                    kept.append(g)
                    continue
                }
                changed = true
                if remaining.isEmpty { continue }
                group["hooks"] = remaining
                kept.append(group)
            }
            if !changed {
                out[event] = value
            } else if !kept.isEmpty {
                out[event] = kept
            }
        }
        return out
    }

    public static func install(into data: Data, cliPath: String) throws -> Data {
        var root = try parse(data)
        var hooks = strip(try hooksDict(root))
        let handler: [String: Any] = ["type": "command", "command": command(cliPath: cliPath), "async": true, "timeout": 5]
        for e in events {
            if let existing = hooks[e], !(existing is [Any]) { throw InstallError.unexpectedShape("hooks.\(e)") }
            var groups = hooks[e] as? [Any] ?? []
            groups.append(["hooks": [handler]])
            hooks[e] = groups
        }
        root["hooks"] = hooks
        return try serialize(root)
    }

    public static func uninstall(from data: Data) throws -> Data {
        var root = try parse(data)
        let hooks = try hooksDict(root)
        let stripped = strip(hooks)
        if stripped.isEmpty && !hooks.isEmpty {
            root.removeValue(forKey: "hooks")
        } else if root["hooks"] != nil {
            root["hooks"] = stripped
        }
        return try serialize(root)
    }

    public static func isInstalled(_ data: Data) -> Bool {
        guard let root = try? parse(data), let hooks = try? hooksDict(root) else { return false }
        return events.allSatisfy { e in
            (hooks[e] as? [Any])?.contains { g in
                ((g as? [String: Any])?["hooks"] as? [Any])?.contains(where: isOurs) ?? false
            } ?? false
        }
    }

    /// Reads `url` (a missing file counts as empty), transforms it, writes a timestamped backup, then
    /// writes the result through any symlink. Nothing is written if `transform` throws.
    /// Returns the backup URL, or nil when there was no file to back up.
    @discardableResult
    public static func apply(to url: URL, now: Date = Date(), transform: (Data) throws -> Data) throws -> URL? {
        let fm = FileManager.default
        let target = url.resolvingSymlinksInPath()
        let exists = fm.fileExists(atPath: target.path)
        let original = exists ? try Data(contentsOf: target) : Data()
        let updated = try transform(original)
        var backup: URL?
        if exists {
            let f = DateFormatter()
            f.locale = Locale(identifier: "en_US_POSIX")
            f.dateFormat = "yyyyMMdd-HHmmss"
            let b = target.deletingLastPathComponent()
                .appendingPathComponent("\(target.lastPathComponent).yogo-backup-\(f.string(from: now))")
            try original.write(to: b)
            backup = b
        } else {
            try fm.createDirectory(at: target.deletingLastPathComponent(), withIntermediateDirectories: true)
        }
        try updated.write(to: target, options: .atomic)
        return backup
    }
}
```

```swift
      yogo hook                                         Claude Code hook'u (JSON'u stdin'den okur)

    """, code: 64)

    // Spec §4: never print, never fail, never block Claude Code. The alarm is a last-resort guard.
    signal(SIGALRM) { _ in _exit(0) }
    alarm(2)
    let input = FileHandle.standardInput.readDataToEndOfFile()
    if let message = SignalMessage.fromHookPayload(input) {

    var session = "simulate"

    for message in SignalMessage.simulate(state, session: session) {

        }
```

```swift
/// The opened 0xFF60/0x61 config interface (spec §3).
final class HIDScreenDevice: ScreenDevice {
    enum HIDError: Error {
        case closed
        case io(IOReturn)
    }

    let ref: IOHIDDevice
    private(set) var isOpen = false

    init(ref: IOHIDDevice) { self.ref = ref }

    func open() -> IOReturn {
        let r = IOHIDDeviceOpen(ref, IOOptionBits(kIOHIDOptionsTypeNone))
        isOpen = r == kIOReturnSuccess
        return r
    }

    func send(_ report: [UInt8]) throws {
        guard isOpen else { throw HIDError.closed }
        var r = IOHIDDeviceSetReport(ref, kIOHIDReportTypeOutput, 0, report, report.count)
        if r != kIOReturnSuccess {   // spec §4: retry once
            r = IOHIDDeviceSetReport(ref, kIOHIDReportTypeOutput, 0, report, report.count)
        }
        guard r == kIOReturnSuccess else { throw HIDError.io(r) }
    }

    func close() {
        guard isOpen else { return }
        isOpen = false
        IOHIDDeviceClose(ref, IOOptionBits(kIOHIDOptionsTypeNone))
    }
}
```

```swift
import AppKit
import IOKit.hid
import ServiceManagement
import YogoCore

/// Owns the whole pipeline (spec §4). Everything runs on the main thread: the 20 Hz timer,
/// IOKit callbacks (main run loop) and the socket handler (main queue).
final class AppModel: ObservableObject {
    static let shared = AppModel()

    enum Connection: String {
        case usb, bluetooth, disconnected, busy
    }

    @Published private(set) var connection: Connection = .disconnected
    @Published private(set) var claudeState: ClaudeState = .idle
    @Published private(set) var sessionCount = 0
    @Published private(set) var animations: [PixelAnimation] = BuiltinAnimations.all
    @Published private(set) var hooksInstalled = false
    @Published var settings = AppSettings.load() {
        didSet {
            settings.save()
            settingsChanged()
        }
    }
    /// Set by the editor while "Klavyede canlı önizle" is on (spec §6, rule 1).
    @Published var previewAnimation: PixelAnimation?

    let tracker = ClaudeTracker()
    let store = AnimationStore(directory: YogoPaths.animationsDir)
    private let driver = ScreenDriver()
    private var selector = SceneSelector()
    private var rejections = RejectionGuard()
    private(set) var assignments = BuiltinAnimations.defaultAssignments
    private var pinned: PixelAnimation?

    private let monitor = HIDMonitor()
    private var device: HIDScreenDevice?
    private var deviceRef: IOHIDDevice?
    private var server: SocketServer?
    private var frameTimer: Timer?
    private var statusTimer: Timer?
    private let log = Log.shared

    // MARK: lifecycle

    func start() {
        store.reload()
        if !store.skippedFiles.isEmpty { log.warn("skipped animation files: \(store.skippedFiles)") }
        animations = store.all
        settingsChanged()
        startServer()
        monitor.onMatch = { [weak self] ref in
            self?.deviceRef = ref
            self?.openDevice()
        }
        monitor.onRemove = { [weak self] ref in self?.deviceRemoved(ref) }
        monitor.onReport = { [weak self] report in self?.handleReply(report) }
        monitor.start()
        observeSleep()
        frameTimer = repeating(Double(Player.tickMs) / 1000) { [weak self] in self?.tick() }
        statusTimer = repeating(5) { [weak self] in self?.periodicCheck() }
        periodicCheck()
        log.info("started")
    }

    func shutdown() {
        frameTimer?.invalidate()
        statusTimer?.invalidate()
        driver.release()
        closeDevice()
        server?.stop()
        log.info("quit")
        log.flush()
    }

    private func repeating(_ interval: TimeInterval, _ body: @escaping () -> Void) -> Timer {
        let t = Timer(timeInterval: interval, repeats: true) { _ in body() }
        RunLoop.main.add(t, forMode: .common)   // keeps running while the menu is open
        return t
    }

    private func startServer() {
        let s = SocketServer(path: YogoPaths.socket, handlerQueue: .main) { [weak self] line in
            guard let self else { return nil }
            return IPCDispatch.handle(line,
                                      onSignal: { self.tracker.apply($0, at: Date()) },
                                      status: { StatusReply(connection: self.connection.rawValue,
                                                            state: self.tracker.state(at: Date()),
                                                            sessions: self.tracker.sessionCount) })
        }
        do {
            try s.start()
            server = s
        } catch SocketServer.ServerError.alreadyRunning {
            let a = NSAlert()
            a.messageText = "Yogo zaten çalışıyor"
            a.informativeText = "Klavyeyi başka bir Yogo örneği kullanıyor. Bu örnek kapanacak."
            a.runModal()
            NSApp.terminate(nil)
        } catch {
            log.warn("socket server failed: \(error)")
        }
    }

    // MARK: device

    private func openDevice() {
        guard device == nil, let ref = deviceRef else {
            refreshConnection()
            return
        }
        let d = HIDScreenDevice(ref: ref)
        let r = d.open()
        if r == kIOReturnSuccess {
            device = d
            driver.attach(d)
            log.info("keyboard opened")
        } else {
            log.warn("keyboard open failed: 0x\(String(UInt32(bitPattern: r), radix: 16))")
        }
        refreshConnection()
    }

    private func closeDevice() {
        driver.attach(nil)
        device?.close()
        device = nil
    }

    private func deviceRemoved(_ ref: IOHIDDevice) {
        guard let current = deviceRef, CFEqual(current, ref) else { return }
        closeDevice()
        deviceRef = nil
        log.info("keyboard removed")
        refreshConnection()
    }

    private func refreshConnection() {
        let c: Connection
        if device != nil {
            c = .usb
        } else if deviceRef != nil {
            c = .busy
        } else if BluetoothProbe.isPresent() {
            c = .bluetooth
        } else {
            c = .disconnected
        }
        if c != connection {
            connection = c
            log.info("connection: \(c.rawValue)")
        }
    }

    /// Every 5 s: retry a busy device, re-check Bluetooth presence and the hooks file.
    private func periodicCheck() {
        if device == nil, deviceRef != nil { openDevice() } else { refreshConnection() }
        let installed = HooksSetup.isInstalled()
        if installed != hooksInstalled { hooksInstalled = installed }
    }

    private func handleReply(_ reply: [UInt8]) {
        let now = Date()
        let wasPaused = rejections.isPaused(at: now)
        rejections.record(reply: reply, at: now)
        if Packet.isRejection(reply) {
            log.warn("rejected: " + reply.prefix(8).map { String(format: "%02x", $0) }.joined(separator: " "))
        }
        if !wasPaused, rejections.isPaused(at: now) { log.warn("5 rejections in a row: live mode paused for 30 s") }
    }

    private func observeSleep() {
        let nc = NSWorkspace.shared.notificationCenter
        nc.addObserver(forName: NSWorkspace.willSleepNotification, object: nil, queue: .main) { [weak self] _ in
            self?.log.info("sleep")
            self?.driver.release()
            self?.closeDevice()
        }
        nc.addObserver(forName: NSWorkspace.didWakeNotification, object: nil, queue: .main) { [weak self] _ in
            self?.log.info("wake")
            self?.openDevice()
        }
    }

    // MARK: 20 Hz loop

    private func tick() {
        let now = Date()
        let state = tracker.state(at: now)
        if state != claudeState { claudeState = state }
        if tracker.sessionCount != sessionCount { sessionCount = tracker.sessionCount }
        let inputs = SceneInputs(preview: previewAnimation,
                                 claudeModeEnabled: settings.claudeModeEnabled,
                                 claudeState: state,
                                 assignments: assignments,
                                 pinned: settings.idleBehavior == .pinned ? pinned : nil)
        let scene = selector.select(inputs, at: now)
        let frame = rejections.isPaused(at: now) ? nil : Player.frame(for: scene, at: now)
        driver.tick(frame: frame, at: now)
        if driver.consecutiveErrors >= 5 {
            log.warn("5 send errors in a row: reopening keyboard")
            closeDevice()
            openDevice()
        }
    }

    // MARK: animations and settings

    private func settingsChanged() {
        assignments = Assignments.resolve(settings.assignments, lookup: store.get)
        pinned = settings.pinnedAnimationID.flatMap(store.get)
        tracker.config.staleTimeout = TimeInterval(settings.staleTimeoutMinutes * 60)
        tracker.config.waitingTimeout = TimeInterval(settings.waitingTimeoutMinutes * 60)
        tracker.config.doneDuration = Double(Player.totalMs(assignedAnimation(for: .done))) / 1000
    }

    func assignedAnimation(for state: ClaudeState) -> PixelAnimation {
        assignments[state] ?? BuiltinAnimations.defaultAssignments[state] ?? BuiltinAnimations.thinking
    }

    func saveAnimation(_ a: PixelAnimation) {
        do {
            try store.save(a)
        } catch {
```

```swift
import CoreGraphics
import Foundation
import ImageIO

/// PNG/GIF/JPEG → 6×6 animation, as a starting point for the editor (spec §8).
public enum ImageImporter {
    public enum ImportError: Error, Equatable {
        case unreadable
        case noFrames
    }

    /// Draw at 48×48 with high-quality interpolation, then box-average 8×8 blocks: a true area filter at 6×6.
    static let workSide = 48

    public static func importAnimation(from url: URL, name: String) throws -> PixelAnimation {
        guard let src = CGImageSourceCreateWithURL(url as CFURL, nil) else { throw ImportError.unreadable }
        let count = CGImageSourceGetCount(src)
        guard count > 0 else { throw ImportError.noFrames }
        let starts = sampleStarts(count: count, limit: PixelAnimation.maxFrames)
        var frames: [AnimationFrame] = []
        for (k, start) in starts.enumerated() {
            guard let image = CGImageSourceCreateImageAtIndex(src, start, nil) else { continue }
            let end = k + 1 < starts.count ? starts[k + 1] : count
            let ms = count == 1 ? 1000 : (start..<end).reduce(0) { $0 + delayMs(src, $1) }
            frames.append(AnimationFrame(durationMs: PixelAnimation.clampDuration(ms), frame: downscale(image)))
        }
        guard !frames.isEmpty else { throw ImportError.noFrames }
        return PixelAnimation(id: "user." + UUID().uuidString.lowercased(), name: name, loop: true, frames: frames)
    }

    /// Evenly spaced source frames; each sampled frame lasts until the next sampled one starts.
    static func sampleStarts(count: Int, limit: Int) -> [Int] {
        count <= limit ? Array(0..<count) : (0..<limit).map { $0 * count / limit }
    }

    static func delayMs(_ src: CGImageSource, _ index: Int) -> Int {
        guard let props = CGImageSourceCopyPropertiesAtIndex(src, index, nil) as? [CFString: Any],
              let gif = props[kCGImagePropertyGIFDictionary] as? [CFString: Any] else { return 100 }
        let s = (gif[kCGImagePropertyGIFUnclampedDelayTime] as? Double) ?? (gif[kCGImagePropertyGIFDelayTime] as? Double) ?? 0.1
        return Int(((s > 0 ? s : 0.1) * 1000).rounded())
    }

    /// Centre-crop to a square, then downscale to 6×6. Premultiplied alpha makes transparent areas "off".
    public static func downscale(_ image: CGImage) -> Frame {
        let side = min(image.width, image.height)
        let crop = CGRect(x: (image.width - side) / 2, y: (image.height - side) / 2, width: side, height: side)
        let square = image.cropping(to: crop) ?? image
        let n = workSide
        let block = workSide / Frame.width
        var buf = [UInt8](repeating: 0, count: n * n * 4)
        buf.withUnsafeMutableBytes { raw in
            guard let ctx = CGContext(data: raw.baseAddress, width: n, height: n, bitsPerComponent: 8, bytesPerRow: n * 4,
                                      space: CGColorSpace(name: CGColorSpace.sRGB)!,
                                      bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return }
            ctx.interpolationQuality = .high
            ctx.draw(square, in: CGRect(x: 0, y: 0, width: n, height: n))
        }
        // Bitmap memory row 0 is the top row of the drawn image.
        var pixels: [RGB] = []
        for by in 0..<Frame.height {
            for bx in 0..<Frame.width {
                var r = 0, g = 0, b = 0
                for y in (by * block)..<((by + 1) * block) {
                    for x in (bx * block)..<((bx + 1) * block) {
                        let i = (y * n + x) * 4
                        r += Int(buf[i]); g += Int(buf[i + 1]); b += Int(buf[i + 2])
                    }
                }
                let area = block * block
                pixels.append(RGB(UInt8(r / area), UInt8(g / area), UInt8(b / area)))
            }
        }
        return Frame(pixels: pixels)
    }
}
```

```swift
/// Pure editing model behind the editor window (spec §9): tools, frames, undo/redo.
/// UI calls `checkpoint()` once per gesture; frame operations checkpoint themselves.
public struct EditorDocument: Equatable {
    public enum Tool: String, CaseIterable, Sendable {
        case pencil, eraser, fill
    }

    public static let undoLimit = 100

    public private(set) var animation: PixelAnimation
    public private(set) var selected = 0
    private var undoStack: [PixelAnimation] = []
    private var redoStack: [PixelAnimation] = []

    public init(animation: PixelAnimation) { self.animation = animation }

    public var currentFrame: Frame { animation.frames[selected].frame }
    public var canUndo: Bool { !undoStack.isEmpty }
    public var canRedo: Bool { !redoStack.isEmpty }

    public mutating func checkpoint() {
        undoStack.append(animation)
        if undoStack.count > Self.undoLimit { undoStack.removeFirst() }
        redoStack.removeAll()
    }

    /// Applies `tool` at (x, y) on the selected frame. Does not checkpoint.
    public mutating func paint(_ tool: Tool, x: Int, y: Int, color: RGB) {
        guard (0..<Frame.width).contains(x), (0..<Frame.height).contains(y) else { return }
        var f = currentFrame
        switch tool {
        case .pencil: f[x, y] = color
        case .eraser: f[x, y] = .off
        case .fill: f = Self.flood(f, x: x, y: y, with: color)
        }
        animation.frames[selected].pixels = f.pixels
    }

    static func flood(_ frame: Frame, x: Int, y: Int, with color: RGB) -> Frame {
        let target = frame[x, y]
        guard target != color else { return frame }
        var f = frame
        var stack = [(x, y)]
        while let p = stack.popLast() {
            let (cx, cy) = p
            guard (0..<Frame.width).contains(cx), (0..<Frame.height).contains(cy), f[cx, cy] == target else { continue }
            f[cx, cy] = color
            stack += [(cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)]
        }
        return f
    }

    public mutating func select(_ index: Int) {
        selected = min(max(index, 0), animation.frames.count - 1)
    }

    /// Inserts a blank frame after the selected one, with the same duration. No-op at the 20-frame limit.
    public mutating func addFrame() {
        guard animation.frames.count < PixelAnimation.maxFrames else { return }
        checkpoint()
        animation.frames.insert(AnimationFrame(durationMs: animation.frames[selected].durationMs, frame: .blank), at: selected + 1)
        selected += 1
    }

    public mutating func duplicateFrame() {
        guard animation.frames.count < PixelAnimation.maxFrames else { return }
        checkpoint()
        animation.frames.insert(animation.frames[selected], at: selected + 1)
        selected += 1
    }

    /// Removes the selected frame; an animation always keeps at least one.
    public mutating func deleteFrame() {
        guard animation.frames.count > 1 else { return }
        checkpoint()
        animation.frames.remove(at: selected)
        selected = min(selected, animation.frames.count - 1)
    }

    public mutating func setDuration(_ ms: Int) {
        let v = PixelAnimation.clampDuration(ms)
        guard v != animation.frames[selected].durationMs else { return }
        checkpoint()
        animation.frames[selected].durationMs = v
    }

    public mutating func setLoop(_ loop: Bool) {
        guard loop != animation.loop else { return }
        checkpoint()
        animation.loop = loop
    }

    /// Renaming is not an undo step (it is typed character by character).
    public mutating func rename(_ name: String) { animation.name = name }

    public mutating func undo() {
        guard let previous = undoStack.popLast() else { return }
        redoStack.append(animation)
        animation = previous
        select(selected)
    }

    public mutating func redo() {
        guard let next = redoStack.popLast() else { return }
        undoStack.append(animation)
        animation = next
        select(selected)
    }

    /// Gives the document a new identity (used when a built-in is forked) across the whole history,
    /// so undo can never bring back a built-in id.
    public mutating func adopt(id: String, name: String) {
        func renamed(_ a: PixelAnimation) -> PixelAnimation {
            var a = a
            a.id = id
            a.name = name
            return a
        }
        animation = renamed(animation)
        undoStack = undoStack.map(renamed)
        redoStack = redoStack.map(renamed)
    }
}
```

```swift
import AppKit
import SwiftUI
```

```swift
final class EditorModel: ObservableObject {
    @Published var doc: EditorDocument
    @Published var tool: EditorDocument.Tool = .pencil
    @Published var color = RGB(hex: "e2794f")!
    @Published var livePreview = false {
        didSet { pushPreview() }
    }
    @Published private(set) var playing = false
    @Published private(set) var playStart = Date()

    private let app: AppModel

    init(app: AppModel) {
        self.app = app
        doc = EditorDocument(animation: app.assignedAnimation(for: .thinking))
    }

    var isBuiltin: Bool { AnimationStore.isBuiltin(doc.animation.id) }

    func open(_ a: PixelAnimation) {
        doc = EditorDocument(animation: a)
        playing = false
        pushPreview()
    }

    func select(_ index: Int) { doc.select(index) }

    // MARK: painting: one drag is one undo step

    func beginStroke() {
        forkIfBuiltin()
        doc.checkpoint()
    }

    func paint(x: Int, y: Int) {
        doc.paint(tool, x: x, y: y, color: color)
        pushPreview()
    }

    func endStroke() { save() }

    /// Structural edits (frames, durations, loop, name) all go through here.
    func edit(_ change: (inout EditorDocument) -> Void) {
        forkIfBuiltin()
        change(&doc)
        save()
        pushPreview()
    }

    func undo() { doc.undo(); save(); pushPreview() }
    func redo() { doc.redo(); save(); pushPreview() }

    func togglePlay() {
        playStart = Date()
        playing.toggle()
    }

    func importImage() {
        let panel = NSOpenPanel()
        panel.allowedContentTypes = [.png, .gif, .jpeg]
        panel.allowsMultipleSelection = false
        NSApp.activate()
        guard panel.runModal() == .OK, let url = panel.url else { return }
        do {
            let a = try ImageImporter.importAnimation(from: url, name: url.deletingPathExtension().lastPathComponent)
            app.saveAnimation(a)
            open(a)
        } catch {
            let alert = NSAlert()
            alert.messageText = "Görsel içe aktarılamadı"
            alert.informativeText = "\(error)"
            alert.runModal()
        }
    }

    func windowClosed() {
        save()
        livePreview = false
        playing = false
        app.previewAnimation = nil
    }

    /// Spec §7: the first edit of a built-in creates a user copy and moves its assignments/pin to it.
    /// Must run before `checkpoint()` so the undo history never holds a built-in id.
    private func forkIfBuiltin() {
        guard isBuiltin else { return }
        let original = doc.animation.id
        let copy = AnimationStore.copy(of: doc.animation)
        doc.adopt(id: copy.id, name: copy.name)
        app.saveAnimation(doc.animation)
        app.moveAssignments(from: original, to: copy.id)
    }

    private func save() {
        if !isBuiltin { app.saveAnimation(doc.animation) }
    }

    private func pushPreview() {
        app.previewAnimation = livePreview ? doc.animation : nil
    }
}
```

```swift
struct CanvasView: View {
    @ObservedObject var model: EditorModel
    @State private var stroking = false
    @State private var lastCell: Int?

    static let cell: CGFloat = 40
    static let gap: CGFloat = 5

    var body: some View {
        TimelineView(.animation(paused: !model.playing)) { context in
            grid(displayed(at: context.date))
        }
        .padding(12)
        .background(RoundedRectangle(cornerRadius: 10).fill(Color(white: 0.04)))
    }

    private func displayed(at date: Date) -> Frame {
        guard model.playing else { return model.doc.currentFrame }
        return Player.frame(for: .play(model.doc.animation, since: model.playStart, loop: true), at: date) ?? model.doc.currentFrame
    }

    private func grid(_ frame: Frame) -> some View {
        VStack(spacing: Self.gap) {
            ForEach(0..<Frame.height, id: \.self) { y in
                HStack(spacing: Self.gap) {
                    ForEach(0..<Frame.width, id: \.self) { x in
                        RoundedRectangle(cornerRadius: 6)
                            .fill(frame[x, y].color)
                            .frame(width: Self.cell, height: Self.cell)
                    }
                }
            }
        }
        .contentShape(Rectangle())
        .gesture(
            DragGesture(minimumDistance: 0)
                .onChanged { value in
                    guard !model.playing else { return }
                    let step = Self.cell + Self.gap
                    let x = Int((value.location.x / step).rounded(.down))
                    let y = Int((value.location.y / step).rounded(.down))
                    guard (0..<Frame.width).contains(x), (0..<Frame.height).contains(y) else { return }
                    if !stroking {
                        stroking = true
                        lastCell = nil
                        model.beginStroke()
                    }
                    let index = y * Frame.width + x
                    guard index != lastCell else { return }
                    lastCell = index
                    model.paint(x: x, y: y)
                }
                .onEnded { _ in
                    guard stroking else { return }
                    stroking = false
                    model.endStroke()
                }
        )
    }
}
```

```swift
                    ForEach(states, id: \.self) { s in
                        Button(s.label) { app.assign(model.doc.animation.id, to: s) }
                    }
                    Divider()
                    Menu("Varsayılana dön") {
                        ForEach(states, id: \.self) { s in
                            Button(s.label) { app.assign(nil, to: s) }
                        }
                    }
                }
                .fixedSize()
                Button("📌 Ekrana sabitle") { app.pin(model.doc.animation.id) }
                    .buttonStyle(.borderedProminent)
            }
            .padding(.horizontal, 12)
            .padding(.bottom, 10)
        }
    }

    private func thumbnail(_ f: Frame) -> some View {
        VStack(spacing: 1) {
            ForEach(0..<Frame.height, id: \.self) { y in
                HStack(spacing: 1) {
                    ForEach(0..<Frame.width, id: \.self) { x in
                        Rectangle().fill(f[x, y].color).frame(width: 7, height: 7)
                    }
                }
            }
        }
        .padding(4)
        .background(RoundedRectangle(cornerRadius: 5).fill(Color(white: 0.04)))
    }
}
```

````markdown
# Yogo

ATK YOGO 75 PRO klavyesinin 6×6 ekranında Claude Code'un durumunu gösteren menü çubuğu uygulaması.

## Kurulum

scripts/build-app.sh --open      # derler, ~/Applications/Yogo.app'e kurar ve açar

## Kullanım

| Claude | Ekran |
|---|---|
| Düşünüyor | Kıvılcım nefes alıyor |
| Araç çalıştırıyor | Artı ve çarpı dönüyor |
| Onay bekliyor | Çift yanıp sönme |
| Bitti | Patlama ve yeşil halka, sonra klavyenin kendi efekti |

- **Terminal:**
  - `~/Applications/Yogo.app/Contents/MacOS/yogo simulate <thinking|tool|waiting|done|idle>`
  - `… yogo status`

## Bilinen kısıtlar

- Canlı ekran yalnızca **USB kabloyla** çalışır. Bluetooth'ta firmware'de ayar kanalı yok, 2.4G dongle ise henüz denenmedi (Faz 2).

- Terminal `SessionEnd` gelmeden kapanırsa: menü → **Claude durumunu sıfırla**.

Yogo klavyeye yalnızca geçici canlı mod komutlarını gönderir: `0x10`, `0x11`, `0x3C` ve `0x3D`. Flash'a yazmaz, firmware arayüzüne dokunmaz. Klavye 0,5 saniye içinde kendi efektine döner.
````
